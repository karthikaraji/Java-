import java.util.Scanner;
public class DecimalToOctal {
static void octal(int n) {
int octal = 0;
int place = 1;
while (n > 0) {
int rem = n % 8;
octal = octal + (rem * place);
place = place * 10;
n = n / 8;
}
System.out.println("Octal = " + octal);
}
public static void main(String args[]) {
Scanner sc = new Scanner(System.in);
int n = sc.nextInt();
octal(n);
    }
}
